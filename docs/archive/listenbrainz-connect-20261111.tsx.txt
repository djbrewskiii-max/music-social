// Preserved developer experiment from src/app/listenbrainz-connect.tsx; not part of launch UI.
import { router } from 'expo-router';
import { useEffect, useState } from 'react';
import { ActivityIndicator, Alert, Pressable, SafeAreaView, ScrollView, StyleSheet,
  Text, TextInput, View } from 'react-native';
import * as WebBrowser from 'expo-web-browser';
import { supabase } from '../supabase';

const CONNECTION='https://listenbrainz.org/settings/music-services/details/';
const usernamePattern=/^[A-Za-z0-9_.-]{2,80}$/;
function getUsername(input:string):string{
  const raw=input.trim();
  if(!raw)return '';
  try{
    if(/^https?:\/\//i.test(raw)){
      const url=new URL(raw);
      if(url.hostname!=='listenbrainz.org'&&url.hostname!=='www.listenbrainz.org')return '';
      return decodeURIComponent(url.pathname.match(/^\/user\/([^/]+)\/?$/)?.[1]||'');
    }
  }catch{return '';}
  return raw.replace(/^@/,'');
}

export default function ListenBrainzConnectScreen(){
  const [username,setUsername]=useState('');
  const [enabled,setEnabled]=useState(false);
  const [busy,setBusy]=useState(false);
  const [loaded,setLoaded]=useState(false);
  useEffect(()=>{
    let active=true;
    void (async()=>{
      const {data:{user}}=await supabase.auth.getUser();
      if(!active||!user)return;
      const {data,error}=await supabase.from('member_listenbrainz_connections')
        .select('listenbrainz_username,sync_enabled').eq('user_id',user.id).maybeSingle();
      if(active&&!error&&data){
        setUsername(data.listenbrainz_username);
        setEnabled(data.sync_enabled===true);
      }
    })().catch(error=>console.log('Listening connection:',error))
      .finally(()=>{if(active)setLoaded(true);});
    return()=>{active=false;};
  },[]);

  const openSetup=async()=>{
    try{await WebBrowser.openBrowserAsync(CONNECTION);}
    catch{Alert.alert('Could not open setup','Visit listenbrainz.org and connect Spotify under Settings → Music services.');}
  };
  const save=async()=>{
    if(busy)return;
    const name=getUsername(username);
    if(!usernamePattern.test(name)){
      Alert.alert('Check your username','Enter the name on your ListenBrainz profile or its public profile link.');
      return;
    }
    setBusy(true);
    try{
      // Validate with the documented public endpoint; a user can have zero
      // listens while Spotify's importer is still starting up.
      const response=await fetch(
        'https://api.listenbrainz.org/1/user/'+encodeURIComponent(name)+'/listen-count',
        {headers:{Accept:'application/json'}}
      );
      if(response.status===404)throw new Error('That ListenBrainz profile was not found. Check the username and try again.');
      if(!response.ok)throw new Error('ListenBrainz is unavailable right now. Please try again.');
      const {data:{user}}=await supabase.auth.getUser();
      if(!user)throw new Error('Sign in to OVRLEE first.');
      // Row ownership is enforced by RLS. Use separate INSERT/UPDATE so
      // the mobile app never needs permission to update a user_id column.
      const {data:existing,error:readError}=await supabase
        .from('member_listenbrainz_connections')
        .select('user_id').eq('user_id',user.id).maybeSingle();
      if(readError)throw readError;
      if(existing){
        const {error}=await supabase.from('member_listenbrainz_connections')
          .update({listenbrainz_username:name,sync_enabled:true,
            updated_at:new Date().toISOString()}).eq('user_id',user.id);
        if(error)throw error;
      }else{
        const {error}=await supabase.from('member_listenbrainz_connections')
          .insert({user_id:user.id,listenbrainz_username:name,sync_enabled:true});
        if(error)throw error;
      }
      setUsername(name);setEnabled(true);
      Alert.alert('Your music stays in sync',
        'OVRLEE will automatically check new listens through ListenBrainz each day. Songs and artists personalize your suggestions privately.',
        [{text:'Done',onPress:()=>router.back()}]);
    }catch(e){
      Alert.alert('Unable to connect',e instanceof Error?e.message:'Try again.');
    }finally{setBusy(false);}
  };
  const disconnect=async()=>{
    if(busy)return;
    setBusy(true);
    try{
      const {data:{user}}=await supabase.auth.getUser();
      if(!user)throw new Error('Sign in first.');
      const {error}=await supabase.from('member_listenbrainz_connections')
        .delete().eq('user_id',user.id);
      if(error)throw error;
      setEnabled(false);setUsername('');
      Alert.alert('Disconnected','OVRLEE stopped checking ListenBrainz and cleared its saved listening signals. Your manually selected favorites remain.');
    }catch(e){Alert.alert('Could not disconnect',e instanceof Error?e.message:'Try again.');}
    finally{setBusy(false);}
  };

  return <SafeAreaView style={s.root}>
    <ScrollView contentContainerStyle={s.content} keyboardShouldPersistTaps="handled">
      <Pressable onPress={()=>router.back()} accessibilityRole="button">
        <Text style={s.back}>‹ Back</Text>
      </Pressable>
      <Text style={s.eyebrow}>OVRLEE × LISTENBRAINZ</Text>
      <Text style={s.heading}>Your music, always current.</Text>
      <Text style={s.body}>Your Spotify listening can quietly improve OVRLEE's recommendations. No recurring imports, no Spotify Premium, and no paid music connection.</Text>

      <View style={s.card}>
        <View style={s.line}><Text style={s.step}>01</Text><Text style={s.stepTitle}>Connect Spotify once</Text></View>
        <Text style={s.sub}>ListenBrainz is a free music-listening service. Sign in or create an account, then authorize Spotify under Music Services. Return here afterward.</Text>
        <Pressable style={s.outlineButton} accessibilityRole="button" onPress={()=>void openSetup()}>
          <Text style={s.outlineText}>Open Spotify connection setup</Text><Text style={s.chevron}>↗</Text>
        </Pressable>
      </View>

      <View style={s.card}>
        <View style={s.line}><Text style={s.step}>02</Text><Text style={s.stepTitle}>Tell OVRLEE which profile is yours</Text></View>
        <Text style={s.sub}>Enter your ListenBrainz username or paste its profile link once. OVRLEE handles updates after that.</Text>
        <TextInput style={s.input} value={username} onChangeText={setUsername}
          placeholder="ListenBrainz username" placeholderTextColor="#747474"
          autoCapitalize="none" autoCorrect={false} accessibilityLabel="ListenBrainz username"/>
        <Pressable style={[s.primary,busy&&s.disabled]} disabled={busy}
          accessibilityRole="button" onPress={()=>void save()}>
          {busy?<ActivityIndicator color="#000"/>:
            <Text style={s.primaryText}>{enabled?'Update connection':'Enable automatic updates'}</Text>}
        </Pressable>
      </View>

      {enabled&&<Pressable style={s.disconnect} disabled={busy} onPress={()=>void disconnect()} accessibilityRole="button">
        <Text style={s.disconnectText}>Disconnect and clear saved listening activity</Text>
      </Pressable>}
      {!loaded&&<ActivityIndicator color="#39E75F"/>}
      <Text style={s.fine}>Your listening data stays private within OVRLEE. ListenBrainz publishes listening records; don't connect if you prefer your listening history not to be public there. OVRLEE never accesses your Spotify password and doesn't display your individual listens to other members. You can stop this sync anytime.</Text>
    </ScrollView>
  </SafeAreaView>;
}

const s=StyleSheet.create({
  root:{flex:1,backgroundColor:'#080808'},
  content:{padding:24,paddingBottom:65,gap:16},
  back:{color:'#ccc',fontSize:16,marginBottom:12},
  eyebrow:{color:'#39E75F',fontSize:10,fontWeight:'900',letterSpacing:2},
  heading:{color:'#fff',fontSize:35,lineHeight:41,fontWeight:'900',marginTop:6},
  body:{color:'#b4b4b4',fontSize:15,lineHeight:23,marginBottom:8},
  card:{borderColor:'#2c2c2c',borderWidth:1,borderRadius:18,padding:18,gap:14,backgroundColor:'#151515'},
  line:{flexDirection:'row',alignItems:'center',gap:10},
  step:{fontSize:11,color:'#39E75F',fontWeight:'900',letterSpacing:1.3},
  stepTitle:{fontSize:17,color:'#fff',fontWeight:'800',flexShrink:1},
  sub:{fontSize:13,color:'#a4a4a4',lineHeight:21},
  outlineButton:{borderColor:'#505050',borderWidth:1,borderRadius:12,
    paddingHorizontal:14,paddingVertical:15,flexDirection:'row',
    alignItems:'center',justifyContent:'space-between'},
  outlineText:{color:'#fff',fontSize:14,fontWeight:'700'},
  chevron:{color:'#39E75F',fontSize:19},
  input:{backgroundColor:'#090909',borderColor:'#444',borderWidth:1,borderRadius:12,
    paddingHorizontal:15,paddingVertical:16,color:'#fff',fontSize:16},
  primary:{backgroundColor:'#39E75F',minHeight:50,borderRadius:12,
    justifyContent:'center',alignItems:'center',padding:15},
  primaryText:{fontSize:15,color:'#000',fontWeight:'900'},
  disabled:{opacity:.55},
  disconnect:{alignItems:'center',padding:14},
  disconnectText:{color:'#999',fontSize:13,textDecorationLine:'underline'},
  fine:{color:'#888',fontSize:12,lineHeight:19,marginTop:8}
});
